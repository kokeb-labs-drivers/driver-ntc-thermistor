318[[nodiscard]] constexpr bool TableValid(std::span<const TablePoint> t) noexcept {
319 const std::size_t rows = t.size();
320 if ((rows < kMinTableRows) || (rows > kMaxTableRows)) {
321 return false;
322 }
323 bool ok = (t[0].resistance_ohm > 0U) && (t[0].temperature_mc > kAbsoluteZeroMc);
324 for (std::size_t i = 1; i < rows; ++i) {
325 const TablePoint& prev = t[i - 1U];
326 const TablePoint& row = t[i];
327 // Two rows whose logarithms agree to Q16 would divide by zero in the
328 // interpolation.
329 ok = ok && (Ln16(prev.resistance_ohm * kFineScale) > Ln16(row.resistance_ohm * kFineScale)) &&
330 (row.resistance_ohm > 0U) && (row.temperature_mc > prev.temperature_mc);
331 }
332 return ok;
333}

345[[nodiscard]] constexpr bool NoShCoefficients(const Curve& c) noexcept {
346 return (c.sh_a_e15 == 0) && (c.sh_b_e15 == 0) && (c.sh_c_e15 == 0);

351inline constexpr int64_t kShLnSquaredQ16 = 32'300'000;
352
356[[nodiscard]] constexpr bool ShValid(const Config& config) noexcept {
357 const Curve& c = config.curve;
358 const bool bounded = (c.sh_a_e15 >= -kMaxShAE15) && (c.sh_a_e15 <= kMaxShAE15) &&
359 (c.sh_b_e15 >= 1) && (c.sh_b_e15 <= kMaxShBE15) &&
360 (c.sh_c_e15 >= -kMaxShCE15) && (c.sh_c_e15 <= kMaxShCE15);
361 const bool unused = (c.r0_ohm == 0U) && (c.b_kelvin == 0U) && (c.b_tol_ppm == 0U);
362 const bool rising = ((c.sh_b_e15 * 65'536) + (3 * c.sh_c_e15 * kShLnSquaredQ16)) > 0;
363 return bounded && unused && rising && WindowDeclared(config);
364}
365
367[[nodiscard]] constexpr bool BetaValid(const Config& config) noexcept {
368 const Curve& c = config.curve;
369 const bool parameters = (c.r0_ohm >= 1U) && (c.b_kelvin >= kMinBetaKelvin) &&
370 (c.b_kelvin <= kMaxBetaKelvin) && (c.b_t2_mc != c.ref_temperature_mc) &&
371 (c.b_t2_mc > kAbsoluteZeroMc);
372 return parameters && WindowDeclared(config) && NoShCoefficients(c);
373}
374
377[[nodiscard]] constexpr bool CurveValid(const Config& config) noexcept {
378 const Curve& c = config.curve;
379 const bool common =
380 (c.r0_ohm <= kMaxR0Ohm) && (c.ref_temperature_mc > kAbsoluteZeroMc) &&
381 (c.r_tol_ppm <= kMaxTolerancePpm) && (c.b_tol_ppm <= kMaxTolerancePpm) &&
382 (c.curve_error_mc <= kMaxCurveErrorMc) &&
383 ((c.b_kelvin == 0U) || ((c.b_kelvin >= kMinBetaKelvin) && (c.b_kelvin <= kMaxBetaKelvin)));
384 if (!common) {
385 return false;
386 }
387 if (!c.table.empty()) {
388 return TableValid(c.table) && NoShCoefficients(c);
389 }
390 return (c.sh_b_e15 != 0) ? ShValid(config) : BetaValid(config);
391}
392
394[[nodiscard]] constexpr bool DividerValid(const Config& config) noexcept {
395 const bool bits = (config.adc_bits >= kMinAdcBits) && (config.adc_bits <= kMaxAdcBits) &&
396 IsEnumerator(config.topology);
397 const bool series = (config.series_ohm >= 1U) && (config.series_ohm <= kMaxSeriesOhm) &&
398 (config.series_tol_ppm <= kMaxTolerancePpm);
399 const bool ratiometric = (config.adc_vref_mv == 0U) && (config.divider_supply_mv == 0U);
400 const bool separate = (config.adc_vref_mv >= 1U) && (config.adc_vref_mv <= kMaxSupplyMv) &&
401 (config.divider_supply_mv >= 1U) &&
402 (config.divider_supply_mv <= kMaxSupplyMv) &&
403 (config.open_margin_codes >= 1U) && (config.short_margin_codes >= 1U);
404 return bits && series && (ratiometric || separate);
405}
406
409[[nodiscard]] constexpr bool SamplingValid(const Config& config) noexcept {
410 const bool samples = (config.sample_count >= 1U) && (config.sample_count <= kMaxSampleCount);
411 const uint64_t margins =
412 static_cast<uint64_t>(config.open_margin_codes) + config.short_margin_codes;
413 const uint32_t top = (uint32_t{1U} << (config.adc_bits & 31U)) - 1U;
414 const bool window = (config.min_temperature_mc < config.max_temperature_mc) &&
415 (config.offset_mc >= -kMaxOffsetMc) && (config.offset_mc <= kMaxOffsetMc);
416 const bool waits = (config.sample_delay_us <= kMaxWaitUs) && (config.settle_us <= kMaxWaitUs);
417 return samples && window && waits && (margins < static_cast<uint64_t>(top));
418}

435[[nodiscard]] constexpr Error Validate(const Config& config) noexcept {
436 const bool ok =
437 detail::CurveValid(config) && detail::DividerValid(config) && detail::SamplingValid(config);
438 return ok ? Error::Ok : Error::InvalidArgument;
