29// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept
30// the exact signature, `noexcept` and the result type; the driver rejects a
31// hook that is declared but does not match (handbook KB-CPP20-007).
32
34template <typename Port>
35concept DeclaresSetBias = requires(Port& port) { port.SetBias(true); };
36
38template <typename Port>
39concept HasSetBias = requires { ExactHook<Port, bool(bool) noexcept>::Take(&Port::SetBias); };
40
42template <typename Port>
43concept DeclaresDelayUs = requires(Port& port) { port.DelayUs(1_u32); };
44
46template <typename Port>
47concept HasDelayUs = requires { ExactHook<Port, void(uint32_t) noexcept>::Take(&Port::DelayUs); };
