57[[nodiscard]] constexpr uint32_t Ln16(uint64_t x) noexcept {
58 if (x == 0U) {
59 return 0U;
60 }
61 const uint32_t whole = static_cast<uint32_t>(std::bit_width(x)) - 1U; // floor(log2 x), 0..63
62 // The mantissa x / 2^whole in [1, 2), as a Q31 number: [2^31, 2^32).
63 uint64_t m = (whole >= 31U) ? (x >> ((whole - 31U) & 63U)) : (x << ((31U - whole) & 63U));
64 uint32_t fraction = 0U;
65 for (uint32_t step = 0U; step < kLnSteps; ++step) {
66 m = (m * m) >> 31U; // the square, back in Q31: [2^31, 2^33)
67 fraction <<= 1U;
68 if (m >= kTwoPow32) {
69 fraction |= 1U;
70 m >>= 1U;
71 }
72 }
73 const uint32_t log2_q17 = (whole << kLnSteps) | fraction;
74 const uint64_t log2_q16 = (static_cast<uint64_t>(log2_q17) + 1U) >> 1U;
75 return static_cast<uint32_t>(((log2_q16 * kLn2Q32) + kTwoPow31) >> 32U);
76}
